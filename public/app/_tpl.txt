async function make() {
  const btn = UI.$("#make"); btn.disabled = true;
  const patches = pieces.map((p, i) => { const o = brief.pieces[i]; if (p.drop) return { key: p.key, drop: true };
    const patch = { key: p.key }; if (p.accountId !== o.accountId) patch.accountId = p.accountId; if (p.format !== o.format) patch.format = p.format; if (p.emotionKey !== o.emotionKey) patch.emotionKey = p.emotionKey;
    if (p.images.count !== o.images.count || p.images.style !== o.images.style) patch.images = { count: p.images.count, style: p.images.style };
    if (JSON.stringify(p.monetize.affiliate) !== JSON.stringify(o.monetize.affiliate)) patch.monetize = { affiliate: p.monetize.affiliate ? { productQuery: p.monetize.affiliate.productQuery, slot: p.monetize.affiliate.slot } : null };
    if (p.schedule.at !== o.schedule.at) patch.schedule = { at: p.schedule.at }; return Object.keys(patch).length > 1 ? patch : null; }).filter(Boolean);
  const body = { briefId: brief.id }; if (patches.length) body.pieces = patches;
  const r = await UI.api("/api/director-confirm", { body }); btn.disabled = false;
  if (!r.ok) {
    if (r.step === "coin_short") return UI.sheet(`<p class="muted" style="margin:0 0 16px">지금 ${r.have ?? brief.coinsLeft}코인이 있고 ${r.need ?? total()}코인이 필요해요.</p><div class="cta"><a class="btn primary" href="/app/coins.html">충전하기</a></div>`, { title: `코인이 ${Math.max(0, (r.need ?? total()) - (r.have ?? brief.coinsLeft))}개 부족해요` });
    return UI.toast(r.error || "만들기를 시작하지 못했어요");
  }
  UI.done("만들기 시작했어요", () => UI.go("/app/pieces.html"));
}
