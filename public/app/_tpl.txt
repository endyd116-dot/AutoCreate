async function submitAdd(body, f, close) {
  const r = await UI.api("/api/accounts-add", { body });
  const pw = f.querySelector('[name="password"],[name="appPassword"]'); if (pw) pw.value = ""; // 🔴 자격은 요청 후 즉시 비운다
  if (!r.ok) {
    if (r.gated) return close(); // 402 plan_limit 은 UI.gate 가 업셀 시트를 열었다
    if (r.step === "limit") { close(); return UI.sheet(`<p class="muted" style="margin:0 0 16px">${UI.esc(r.error)}</p><div class="cta"><a class="btn primary" href="/app/plan.html">요금제 보기</a></div>`, { title: "이 요금제에서는 여기까지예요" }); }
    if (r.step === "duplicate" || r.step === "creds" || r.step === "handle_policy") return UI.fieldError(f, "loginId", r.error);
    if (r.step === "wp_auth") return UI.fieldError(f, "appPassword", r.error);
    return UI.toast(r.error || "연결하지 못했어요");
  }
  close(); UI.done(r.account?.status === "active" ? "연결했어요" : "연결했어요 · 첫 발행 전에 로그인을 확인해요", load);
}

/* ── 다시 로그인(§7.2 · 내 PC 프로그램이 로그인 창을 연다) ── */
const RELOGIN_STEPS = [["내 PC 프로그램에 부탁해요", "queued"], ["PC에서 로그인 창이 열려요", "running"], ["로그인을 마치면 바로 이어서 올려요", "done"]];
function reloginSteps(status) {
  const i = Math.max(0, RELOGIN_STEPS.findIndex(([, k]) => k === status));
  return `<div class="steps">${RELOGIN_STEPS.map(([l], n) => `<div class="st ${status === "done" || n < i ? "done" : status && n === i ? "on" : ""}"><i>${status === "done" || n < i ? "✓" : n + 1}</i>${l}</div>`).join("")}</div>`;
}

    <p class="muted" style="margin:0 0 8px;font-size:13px">보안 문자가 나오면 그 창에서 직접 입력해 주세요. 비밀번호는 저장된 것을 씁니다.</p>
    <div id="rst">${reloginSteps(null)}</div><div id="rerr"></div>
    <div class="cta" id="rcta"><button class="btn primary" type="button" id="go">로그인 창 열기</button></div>`,
    { title: "다시 로그인", onOpen: (sh, close) => {
      const st = sh.querySelector("#rst"), box = sh.querySelector("#rerr"), cta = sh.querySelector("#rcta");

        e.target.disabled = true;
        const r = await UI.api("/api/accounts-relogin", { body: { id: a.id } });
        if (!r.ok && r.step === "runner_offline") { e.target.disabled = false; box.innerHTML = `<div class="banner" style="margin:8px 0 0"><span>${UI.esc(r.error)}</span><a href="/app/runner.html">켜러 가기</a></div>`; return; }
        cta.hidden = true;
        let status = r.ok && r.job ? r.job.status : null;
        st.innerHTML = reloginSteps(status || "queued");
        const t0 = Date.now();
        const tick = async () => {
          if (!document.body.contains(sh)) return;                       // 시트 닫히면 폴링도 끝
          const g = await UI.api(`/api/accounts-relogin?id=${a.id}`);
          const acc = g.ok ? g.account : null;
          status = g.ok && g.job ? g.job.status : status;
          if (!g.ok || (!g.job && acc)) {                                 // 폴백: 잡을 못 보면 계정 상태로만 판정
            const al = await UI.api("/api/accounts-list");
            const me2 = al.ok ? (al.accounts || []).find((x) => x.id === a.id) : null;
            if (me2 && me2.status === "active") status = "done";
          } else if (acc && acc.status === "active") status = "done";
          st.innerHTML = reloginSteps(status === "failed" ? "running" : status || "queued");
          if (status === "done") { close(); UI.done(`@${a.handle} 다시 로그인했어요`, load); return; }
          if (status === "failed") { box.innerHTML = `<div class="banner" style="margin:8px 0 0"><span>로그인 창에서 끝내지 못했어요</span></div>`; cta.hidden = false; e.target.disabled = false; return; }
          if (Date.now() - t0 > 120000) { box.innerHTML = `<div class="banner" style="margin:8px 0 0"><span>아직 끝나지 않았어요 · PC 화면을 확인해 주세요</span></div>`; cta.hidden = false; e.target.disabled = false; return; }
          setTimeout(tick, 2000);
        };
        setTimeout(tick, 1500);
      };
    } });
}

  if (r.step === "provider_not_configured") return UI.sheet(`<p style="margin:0 0 4px;font-size:16px;font-weight:700">준비가 아직이에요</p><p class="muted" style="margin:0 0 16px">${UI.esc(UI.srcLabel(source))} 구글 연결이 열리면 홈 «해야 할 일»에 알려드릴게요.</p><div class="cta"><button class="btn secondary" type="button" id="ok">알겠어요</button></div>`, { title: "", onOpen: (sh, close) => { sh.querySelector("#ok").onclick = close; } });
  UI.toast(r.error || "연결을 시작하지 못했어요");
}
/* 키 입력 = 바텀시트 · 소스별 여러 칸 · creds 객체로 보낸다(v3.5) */
function keySheet(key) {
  const fields = KEYED[key] || [["key", "키"]]; const url = linkOf(key);
  UI.sheet(`<p class="muted" style="margin:0 0 8px">키는 암호로 잠가서 보관하고, 수익을 가져올 때만 써요.</p>
    ${fields.map(([n, l]) => { const opt = n.endsWith("?"); const name = n.replace(/\?$/, ""); return `<div class="field"><label>${UI.esc(l)}${opt ? ' <span class="muted" style="font-weight:500">선택</span>' : ""}</label><input class="input" data-f="${name}" ${opt ? 'data-opt="1"' : ""} autocomplete="off" type="password"><div class="help"></div></div>`; }).join("")}
    ${url ? `<p class="muted" style="margin:0 0 4px;font-size:12.5px">키는 <a href="${UI.esc(url)}" target="_blank" rel="noopener" style="text-decoration:underline">${UI.esc(UI.srcLabel(key))}</a> 에서 발급받아요.</p>` : ""}
    <div class="cta"><button class="btn primary" type="button" id="ok">연결하기</button></div>`,
    { title: `${UI.srcLabel(key)} 키`, onOpen: (sh, close) => {
      const inputs = [...sh.querySelectorAll("[data-f]")];
      const errAt = (name, msg) => { const i = inputs.find((x) => x.dataset.f === name) || inputs[0]; i.classList.add("err"); i.closest(".field").querySelector(".help").textContent = msg; };
      sh.querySelector("#ok").onclick = async (e) => {
        inputs.forEach((i) => { i.classList.remove("err"); i.closest(".field").querySelector(".help").textContent = ""; });
        const creds = {}; for (const i of inputs) { const v = i.value.trim(); if (!v && !i.dataset.opt) return errAt(i.dataset.f, "키를 모두 넣어 주세요."); if (v) creds[i.dataset.f] = v; }
        e.target.disabled = true;
        const r = await UI.api("/api/revenue-sources", { body: { action: "key", source: key, creds } });
        inputs.forEach((i) => i.value = ""); // 🔴 키는 요청 후 즉시 비운다
        e.target.disabled = false;
        if (!r.ok) { const m = /\(([A-Za-z]+)\)/.exec(r.error || ""); return errAt(m ? m[1] : inputs[0].dataset.f, r.error || "연결하지 못했어요"); } // «키를 모두 넣어 주세요(appSecret)» → 그 칸에
        close(); UI.done("연결했어요", load);
      };
    } });
}
async function disconnect(key, accountId) {
  UI.sheet(`<div id="dc"></div>`, { title: `${UI.srcLabel(key)} 끊기`, onOpen: (sh, close) => {
    UI.confirmRow(sh.querySelector("#dc"), `${UI.srcLabel(key)} 연결을 끊을까요? 지금까지 모은 수익은 그대로 남아요.`, async () => {
      const body = { action: "disconnect", source: key }; if (accountId) body.accountId = accountId;
      const r = await UI.api("/api/revenue-sources", { body });
      if (!r.ok) return UI.toast(r.error || "끊지 못했어요");
      close(); UI.toast("연결을 끊었어요"); load();
    });
  } });
}
